cmake_minimum_required(VERSION 3.22)
project(clawbench_tunnel_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TUNNEL_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../entry/src/main/cpp")
if(WIN32)
  message(FATAL_ERROR "Use Linux or an OHOS toolchain for the POSIX tunnel tests")
endif()

set(TUNNEL_DEPENDENCY_STAGE "" CACHE PATH "Optional prefix containing the tunnel's static dependencies")
set(TUNNEL_JSON_INCLUDE "" CACHE PATH "Optional directory containing nlohmann/json.hpp")
option(TUNNEL_BUILD_DEPENDENCIES "Build the same pinned dependencies as the HarmonyOS application" OFF)
if(TUNNEL_BUILD_DEPENDENCIES)
  include("${TUNNEL_CORE}/ThirdParty.cmake")
  set(TUNNEL_SSH_INCLUDE "${TUNNEL_STAGE}/include")
  set(TUNNEL_H2_INCLUDE "${TUNNEL_STAGE}/include")
  set(TUNNEL_TLS_INCLUDE "${TUNNEL_STAGE}/include")
  set(TUNNEL_JSON_INCLUDE "${tunnel_json_SOURCE_DIR}/single_include")
  set(TUNNEL_TEST_LIBRARIES ${TUNNEL_LIBRARIES})
else()
  if(TUNNEL_DEPENDENCY_STAGE)
    list(PREPEND CMAKE_PREFIX_PATH "${TUNNEL_DEPENDENCY_STAGE}")
  endif()
  find_path(TUNNEL_SSH_INCLUDE libssh2.h HINTS "${TUNNEL_DEPENDENCY_STAGE}/include" NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  find_path(TUNNEL_H2_INCLUDE nghttp2/nghttp2.h HINTS "${TUNNEL_DEPENDENCY_STAGE}/include" NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  find_path(TUNNEL_TLS_INCLUDE mbedtls/ssl.h HINTS "${TUNNEL_DEPENDENCY_STAGE}/include" NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  if(NOT TUNNEL_JSON_INCLUDE)
    unset(TUNNEL_JSON_INCLUDE CACHE)
    find_path(TUNNEL_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
  endif()
  set(TUNNEL_TEST_LIBRARIES "")
  foreach(DEPENDENCY ssh2 mbedtls mbedx509 mbedcrypto nghttp2)
    find_library(TUNNEL_${DEPENDENCY}_LIBRARY NAMES ${DEPENDENCY}
      HINTS "${TUNNEL_DEPENDENCY_STAGE}/lib" NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    list(APPEND TUNNEL_TEST_LIBRARIES "${TUNNEL_${DEPENDENCY}_LIBRARY}")
  endforeach()
  if(TUNNEL_DEPENDENCY_STAGE)
    foreach(DEPENDENCY everest p256m)
      find_library(TUNNEL_${DEPENDENCY}_LIBRARY NAMES ${DEPENDENCY}
        PATHS "${TUNNEL_DEPENDENCY_STAGE}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
      if(TUNNEL_${DEPENDENCY}_LIBRARY)
        list(APPEND TUNNEL_TEST_LIBRARIES "${TUNNEL_${DEPENDENCY}_LIBRARY}")
      endif()
    endforeach()
  endif()
endif()
find_package(Threads REQUIRED)

# The tests use socket pairs, so no certificate bundle or external server is needed.
set(TUNNEL_CA_TEXT "")
configure_file("${TUNNEL_CORE}/trusted_ca.h.in" "${CMAKE_CURRENT_BINARY_DIR}/trusted_ca.h" @ONLY)
add_executable(tunnel-core-test tunnel-core-test.cpp)
if(TUNNEL_BUILD_DEPENDENCIES)
  add_dependencies(tunnel-core-test tunnel_nghttp2 tunnel_ssh2)
endif()
target_include_directories(tunnel-core-test PRIVATE
  "${TUNNEL_CORE}" "${CMAKE_CURRENT_BINARY_DIR}" "${TUNNEL_SSH_INCLUDE}"
  "${TUNNEL_H2_INCLUDE}" "${TUNNEL_TLS_INCLUDE}" "${TUNNEL_JSON_INCLUDE}")
target_compile_options(tunnel-core-test PRIVATE -Wall -Wextra -UNDEBUG)
target_compile_definitions(tunnel-core-test PRIVATE LIBSSH2_LIBRARY NGHTTP2_STATICLIB)
target_link_libraries(tunnel-core-test PRIVATE ${TUNNEL_TEST_LIBRARIES} Threads::Threads)
enable_testing()
add_test(NAME tunnel-core COMMAND tunnel-core-test)
